+        async function submitCredentials() {
+            const adapter = activeAdapter;
+            const keys = {};
+            
+            if (adapter === 'alpaca') {
+                keys['ALPACA_API_KEY'] = document.getElementById('key-alpaca-id').value;
+                keys['ALPACA_API_SECRET'] = document.getElementById('key-alpaca-secret').value;
+            } else if (adapter === 'polygon') {
+                keys['POLYGON_API_KEY'] = document.getElementById('key-polygon-key').value;
+            } else if (adapter === 'oanda') {
+                keys['OANDA_API_KEY'] = document.getElementById('key-oanda-token').value;
+                keys['OANDA_ACCOUNT_ID'] = document.getElementById('key-oanda-account').value;
+            } else if (adapter === 'binance' || adapter === 'binance_futures') {
+                keys['BINANCE_API_KEY'] = document.getElementById('key-binance-key').value;
+                keys['BINANCE_API_SECRET'] = document.getElementById('key-binance-secret').value;
+            } else if (adapter === 'deribit') {
+                keys['DERIBIT_CLIENT_ID'] = document.getElementById('key-deribit-id').value;
+                keys['DERIBIT_CLIENT_SECRET'] = document.getElementById('key-deribit-secret').value;
+            } else if (adapter === 'ib') {
+                keys['IB_HOST'] = document.getElementById('key-ib-host').value;
+                keys['IB_PORT'] = document.getElementById('key-ib-port').value;
+                keys['IB_CLIENT_ID'] = document.getElementById('key-ib-client').value;
+            }
+            
+            // Save to localStorage for browser reload pre-population
+            for (const [k, v] of Object.entries(keys)) {
+                localStorage.setItem(k, v);
+            }
+            
+            closeCredentialsModal();
+            showActionToast('Сохранение ключей на сервере...');
+            
+            try {
+                const res = await fetch(API_BASE + '/api/save_credentials', {
+                    method: 'POST',
+                    headers: {'Content-Type': 'application/json'},
+                    body: JSON.stringify({
+                        adapter: adapter,
+                        keys: keys
+                    })
+                });
+                
+                if (res.ok) {
+                    showActionToast('✅ API Ключи успешно применены и сохранены в .env!');
+                } else {
+                    const errText = await res.text();
+                    showActionToast(`❌ Ошибка сохранения: ${errText}`);
+                }
+            } catch (err) {
+                showActionToast(`❌ Сетевая ошибка: ${err.message}`);
+            }
+        }
